### S-BE-SEC-002: Replace Broken HTML Sanitizer with `bleach`

**Issue:** BE-SEC-002 — Regex sanitizer doesn't work due to double-escaping

**Approach:**
1. Install `bleach`: add `bleach>=6.0` to `requirements.txt`
2. Replace entire `sanitize_html()` function in `backend/app/utils/validators.py` with:

```python
import bleach

ALLOWED_TAGS = [
    "p", "br", "strong", "em", "u", "s", "h1", "h2", "h3", "h4", "h5", "h6",
    "ul", "ol", "li", "a", "img", "blockquote", "pre", "code", "table",
    "thead", "tbody", "tr", "th", "td", "hr", "span", "div", "figure", "figcaption",
]

ALLOWED_ATTRIBUTES = {
    "a": ["href", "title", "target", "rel"],
    "img": ["src", "alt", "width", "height"],
    "span": ["class"],
    "div": ["class"],
    "*": ["class"],
}

ALLOWED_PROTOCOLS = ["http", "https", "mailto"]

def sanitize_html(html: str) -> str:
    return bleach.clean(
        html,
        tags=ALLOWED_TAGS,
        attributes=ALLOWED_ATTRIBUTES,
        protocols=ALLOWED_PROTOCOLS,
        strip=True,
    )
```

**Effort:** Quick Fix  
**Priority:** P0 — This is the root cause of the XSS attack chain.

### S-BE-SEC-003: Add SSRF Protection to URL Validation

**Issue:** BE-SEC-003 — URL validation allows internal networks

**Approach:**  
Add hostname validation to `_validate_url()`:

```python
from urllib.parse import urlparse
import ipaddress
import socket

BLOCKED_HOSTNAMES = {"localhost", "127.0.0.1", "0.0.0.0", "::1"}

def _validate_url(url: Any, field_name: str) -> None:
    if not isinstance(url, str):
        raise ServiceError("VALIDATION_ERROR", f"{field_name} must be a valid URL", 400)
    parsed = urlparse(url)
    if parsed.scheme not in ("http", "https"):
        raise ServiceError("VALIDATION_ERROR", f"{field_name} must use http or https", 400)
    hostname = parsed.hostname
    if not hostname or hostname in BLOCKED_HOSTNAMES:
        raise ServiceError("VALIDATION_ERROR", f"{field_name} contains blocked host", 400)
    try:
        ip = ipaddress.ip_address(hostname)
        if ip.is_private or ip.is_loopback or ip.is_link_local:
            raise ServiceError("VALIDATION_ERROR", f"{field_name} targets private network", 400)
    except ValueError:
        pass  # Hostname is not an IP; that's fine
```

**Effort:** Moderate  
**Priority:** P1

```typescript
// web/lib/sanitize.ts
export function safeHref(url: string | undefined | null): string {
    if (!url) return '#';
    if (url.startsWith('/') || url.startsWith('https://') || url.startsWith('http://')) {
        return url;
    }
    return '#'; // Block javascript:, data:, vbscript:, etc.
}
```
